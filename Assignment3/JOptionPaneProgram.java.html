import javax.swing.JOptionPane;

public class JOptionPaneProgram {
    public static void main(String[] args) {

        double nsat = Double.parseDouble(JOptionPane.showInputDialog("Enter NSAT score: "));

        double salary = Double.parseDouble(JOptionPane.showInputDialog("Enter NSAT score: "));

        double entrance = Double.parseDouble(JOptionPane.showInputDialog("Enter parents' monthly salary: "));

        double average = (nsat + entrance) / 2;

        if (salary > 10000 || nsat < 90 || entrance < 85) {
            JOptionPane.showMessageDialog(null, "Application Status: REJECTED ");
        }

        else if (salary <= 3500 && average >= 91) {
            JOptionPane.showMessageDialog(null, "Application Status: ACCEPTED ");
        }
        else {
            JOptionPane.showMessageDialog(null, "Application Status: FOR FURTHER STUDY");

        }
    }
}
